3．RunBuild の末尾へ「自動算出シート出力＋差分メモ追記」コードを追加

vb
Copy
Public Function RunBuild(Optional ByVal silent As Boolean = False) As String
    ' （中略）既存の RunBuild 全体

    ' 最後に以下を追記してください。

    ' --- 自動算出標準値を「元データから算出」シート(SH_SAVE)へ出力
    Dim wsMap As Worksheet, wsCalc As Worksheet
    Dim lastMap As Long, r As Long
    Set wsMap  = ThisWorkbook.Worksheets(SH_MAP)
    Set wsCalc = ThisWorkbook.Worksheets(SH_SAVE)

    ' シートクリア＆ヘッダ
    With wsCalc
      .Cells.Clear
      .Range("A1:C1").Value = Array("業務ID", "工程×大分類", "自動算出標準（件/時）")
    End With

    ' マスタ_業務・標準の行をループして、平均値を計算・出力
    lastMap = LastRow(wsMap, MC_ID)
    For r = 2 To lastMap
      Dim id As String, stg As String, mt As String, key As String
      Dim totalCnt As Double, totalHrs As Double, avgRate As Double

      id  = wsMap.Cells(r, MC_ID).Value
      stg = wsMap.Cells(r, MC_STAGE).Value
      mt  = wsMap.Cells(r, MC_MT).Value
      key = stg & "|" & mt

      ' dayAgg と所要時間集計：RunBuild 内で定義済みの dayAgg, ds, etc. を想定
      totalCnt = 0: totalHrs = 0
      ' ds: 集計対象日付の配列、dayAgg(k)=(件数, 時間)
      For Each d In ds
        Dim arr As Variant
        If dayAgg.Exists(id & "|" & d & "|" & key) Then
          arr = dayAgg(id & "|" & d & "|" & key)
          totalCnt = totalCnt + arr(0)
          totalHrs = totalHrs + arr(1)
        End If
      Next
      If totalHrs > 0 Then
        avgRate = totalCnt / totalHrs
      Else
        avgRate = 0
      End If

      wsCalc.Cells(r, 1).Value = id
      wsCalc.Cells(r, 2).Value = key
      wsCalc.Cells(r, 3).Value = Round(avgRate, 2)
    Next

    ' --- マスタ登録値（MC_STD）と自動算出値（wsCalc列3）を比較、
    '     差があればマスタ_業務・標準のメモ列(MC_MEMO)に追記
    For r = 2 To lastMap
      Dim manualStd As Variant, autoStd As Variant
      manualStd = wsMap.Cells(r, MC_STD).Value
      autoStd   = wsCalc.Cells(r, 3).Value
      If manualStd <> autoStd Then
        wsMap.Cells(r, MC_MEMO).Value = _
          wsMap.Cells(r, MC_MEMO).Value & _
          "【自動基準:" & autoStd & "との差異】"
      End If
    Next

    '（既存の Notify や終了処理など）
End Function
――――――――――――
以上を適用すれば、

マスタ_業務・標準の H 列（MC_STD）は上書きされず管理者入力値を保持
自動算出した平均標準値は「元データから算出」シートに出力
両者の差分はマスタのメモ列に自動で追記
という振る舞いになります。コピー＆ペーストで該当部分を差し替え、動作をご確認ください。
